#!/bin/sh
S=${S:?directory holding bin5/ with aircd and airc-gateway}
R=$(cd "$(dirname "$0")" && pwd)
rm -rf $R/state; mkdir -p $R/state/gw; chmod 700 $R/state $R/state/gw
AIRC=127.0.0.1:16694; MQTT=16695
python3 -c "import secrets; open('$R/state/gateway.token','w').write(secrets.token_hex(32)+'\n')"; chmod 600 $R/state/gateway.token
$S/bin5/airc-gateway credential init --file $R/state/cursor.key >/dev/null
cat > $R/state/gateway.json <<JSON
{"version":1,"realm":"mqtt5","state_dir":"$R/state/gw","cursor_key_file":"$R/state/cursor.key",
 "upstream":{"network":"tcp","address":"$AIRC","gateway_token_file":"$R/state/gateway.token"},
 "listeners":[{"id":"mqtt","protocol":"mqtt","enabled":true,"address":"127.0.0.1:$MQTT"}]}
JSON
$S/bin5/aircd --listen $AIRC --history 1000 --accounts-file $R/state/accounts.json --gateway-token-file $R/state/gateway.token --log-file $R/state/aircd.log &
AIRCD=$!; sleep 1
$S/bin5/airc-gateway serve --file $R/state/gateway.json > $R/state/gateway.log 2>&1 &
GATEWAY=$!; sleep 1
head -c 200 $R/state/gateway.log; echo
TOKEN=$(python3 -c "import secrets; print(secrets.token_hex(32))")
python3 - <<PY
import socket,time
s=socket.create_connection(("127.0.0.1",16694)); s.sendall(b"REGISTER carol :$TOKEN\r\n"); time.sleep(0.5); print(s.recv(4096).decode().strip()[:80]); s.close()
PY
native() { python3 - "$1" <<'PY'
import socket,sys,time
s=socket.create_connection(("127.0.0.1",16694)); s.sendall(b"EPHEMERAL\r\nNICK native-poster\r\nUSER u 0 * :u\r\n"); time.sleep(0.3)
s.sendall(("PRIVMSG #lobby :"+sys.argv[1]+"\r\n").encode()); time.sleep(0.3); s.close()
PY
}
echo "== mosquitto_rr (guest, request/response)"
mosquitto_rr -V 5 -h 127.0.0.1 -p $MQTT -q 1 -t airc/requests -e replies/rr-guest -W 5 \
  -D connect user-property airc-nick rr-guest \
  -m '{"version":1,"request_id":"rr-1","operation":"channels.read","author":{"nick":"rr-guest"}}' | cut -c1-200
echo "== mosquitto_pub (account carol, a send at QoS 1)"
mosquitto_pub -V 5 -h 127.0.0.1 -p $MQTT -q 1 -t airc/requests -d \
  -D connect user-property airc-nick carol -D connect user-property airc-token $TOKEN \
  -m '{"version":1,"request_id":"pub-1","operation":"send","target":{"kind":"room","name":"#lobby"},"message":"hello from mosquitto_pub as carol"}' 2>&1 | grep -i "puback\|connack\|error" | head -4
echo "== mosquitto_sub (account carol, persistent session)"
mosquitto_sub -V 5 -h 127.0.0.1 -p $MQTT -c -i sub-carol -x 600 -q 1 -t airc/room/lobby -C 1 -W 10 -F '%t %p' \
  -D connect user-property airc-nick carol -D connect user-property airc-token $TOKEN > $R/state/sub1.out 2>&1 &
SUB=$!; sleep 1.5
native "for the subscriber"
wait $SUB; echo "first connection:"; cut -c1-200 $R/state/sub1.out
native "while the subscriber was away 1"
native "while the subscriber was away 2"
mosquitto_sub -V 5 -h 127.0.0.1 -p $MQTT -c -i sub-carol -x 600 -q 1 -t airc/room/lobby -C 2 -W 10 -F '%t %p' \
  -D connect user-property airc-nick carol -D connect user-property airc-token $TOKEN 2>&1 | cut -c1-200 | sed 's/^/resumed: /'
echo "== paho-mqtt"
python3 $R/paho_client.py.txt 127.0.0.1 $MQTT paho-guest 127.0.0.1:16694
echo "== native history"
python3 - <<PY
import socket,time,json,base64
s=socket.create_connection(("127.0.0.1",16694)); s.sendall(b"EPHEMERAL\r\nNICK auditor\r\nUSER u 0 * :u\r\n"); time.sleep(0.5); s.recv(65536)
s.sendall(b"HISTORY #lobby 20\r\n"); time.sleep(0.5)
for line in s.recv(65536).decode().split("\r\n"):
    if " 760 " in line:
        m=json.loads(line.split(" :",1)[1]); print(m["from"], m.get("account_id","-")[:8], base64.b64decode(m["body_base64"]+"==").decode())
PY
kill $GATEWAY $AIRCD 2>/dev/null; wait 2>/dev/null
