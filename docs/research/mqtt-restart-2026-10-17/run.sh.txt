#!/bin/sh
# A mosquitto_sub persistent session across a gateway restart.
S=${S:?directory holding bin6/ with aircd and airc-gateway}
R=$(cd "$(dirname "$0")" && pwd)
rm -rf $R/state; mkdir -p $R/state/gw; chmod 700 $R/state $R/state/gw
AIRC=127.0.0.1:16700; MQTT=16701
python3 -c "import secrets; open('$R/state/gateway.token','w').write(secrets.token_hex(32)+'\n')"; chmod 600 $R/state/gateway.token
$S/bin6/airc-gateway credential init --file $R/state/cursor.key >/dev/null
cat > $R/state/gateway.json <<JSON
{"version":1,"realm":"durable","state_dir":"$R/state/gw","cursor_key_file":"$R/state/cursor.key",
 "upstream":{"network":"tcp","address":"$AIRC","gateway_token_file":"$R/state/gateway.token"},
 "listeners":[{"id":"mqtt","protocol":"mqtt","enabled":true,"address":"127.0.0.1:$MQTT"}]}
JSON
$S/bin6/aircd --listen $AIRC --history 1000 --accounts-file $R/state/accounts.json --gateway-token-file $R/state/gateway.token --log-file $R/state/aircd.log &
AIRCD=$!; sleep 1
gateway() { $S/bin6/airc-gateway serve --file $R/state/gateway.json >> $R/state/gateway.log 2>&1 & GATEWAY=$!; sleep 1; }
gateway
TOKEN=$(python3 -c "import secrets; print(secrets.token_hex(32))")
python3 - <<PY
import socket,time
s=socket.create_connection(("127.0.0.1",16700)); s.sendall(b"REGISTER carol :$TOKEN\r\n"); time.sleep(0.5); print(s.recv(4096).decode().strip()[:80]); s.close()
PY
native() { python3 - "$1" <<'PY'
import socket,sys,time
s=socket.create_connection(("127.0.0.1",16700)); s.sendall(b"EPHEMERAL\r\nNICK native-poster\r\nUSER u 0 * :u\r\n"); time.sleep(0.3)
s.sendall(("PRIVMSG #lobby :"+sys.argv[1]+"\r\n").encode()); time.sleep(0.3); s.close()
PY
}
sub() { mosquitto_sub -V 5 -h 127.0.0.1 -p $MQTT -c -i durable-carol -x 600 -q 1 -t airc/room/lobby -C $1 -W 10 -F '%t %p' \
  -D connect user-property airc-nick carol -D connect user-property airc-token $TOKEN 2>&1 | python3 -c "
import sys,json,base64
for line in sys.stdin:
    topic,_,payload=line.strip().partition(' ')
    try: print(topic, base64.b64decode(json.loads(payload)['body_base64']+'==').decode())
    except Exception: print(line.strip())"; }
echo "== first connection: a persistent session gets a live post"
sub 1 > $R/state/first.out & SUB=$!; sleep 1.5
native "before the restart"
wait $SUB; cat $R/state/first.out
echo "== gateway stopped (SIGTERM)"
kill -TERM $GATEWAY; wait $GATEWAY 2>/dev/null
ls -l $R/state/gw | awk '{print $1, $NF}' | grep -v total
native "while the gateway was down 1"
native "while the gateway was down 2"
echo "== gateway restarted; the session resumes"
gateway
sub 2
kill $GATEWAY $AIRCD 2>/dev/null; wait 2>/dev/null
