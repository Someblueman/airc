#!/bin/sh
# MQTT 5 wills and MQTT over WebSocket against a scratch aircd and gateway.
S=${S:?directory holding binws/ with aircd, airc-gateway and airc}
R=$(cd "$(dirname "$0")" && pwd)
rm -rf $R/state; mkdir -p $R/state/gw; chmod 700 $R/state $R/state/gw
AIRC=127.0.0.1:16696; MQTT=16697; MQTTWS=16698; PAGE=16699
python3 -c "import secrets; open('$R/state/gateway.token','w').write(secrets.token_hex(32)+'\n')"; chmod 600 $R/state/gateway.token
$S/binws/airc-gateway credential init --file $R/state/cursor.key >/dev/null
cat > $R/state/gateway.json <<JSON
{"version":1,"realm":"mqttws","state_dir":"$R/state/gw","cursor_key_file":"$R/state/cursor.key",
 "upstream":{"network":"tcp","address":"$AIRC","gateway_token_file":"$R/state/gateway.token"},
 "listeners":[{"id":"mqtt","protocol":"mqtt","enabled":true,"address":"127.0.0.1:$MQTT"},
  {"id":"mqttws","protocol":"mqtt_websocket","enabled":true,"address":"127.0.0.1:$MQTTWS","allowed_origins":["http://127.0.0.1:$PAGE","http://127.0.0.1:$MQTTWS"]}]}
JSON
$S/binws/aircd --listen $AIRC --history 1000 --accounts-file $R/state/accounts.json --gateway-token-file $R/state/gateway.token --log-file $R/state/aircd.log &
AIRCD=$!; sleep 1
$S/binws/airc-gateway serve --file $R/state/gateway.json > $R/state/gateway.log 2>&1 &
GATEWAY=$!
(cd $R && python3 -m http.server $PAGE --bind 127.0.0.1 > $R/state/page.log 2>&1) &
PAGESERVER=$!; sleep 1
head -c 300 $R/state/gateway.log; echo
TOKEN=$(python3 -c "import secrets; print(secrets.token_hex(32))")
python3 - <<PY
import socket,time
s=socket.create_connection(("127.0.0.1",16696)); s.sendall(b"REGISTER carol :$TOKEN\r\n"); time.sleep(0.5); print(s.recv(4096).decode().strip()[:80]); s.close()
PY
native() { python3 - "$1" "$2" <<'PY'
import socket,sys,time
s=socket.create_connection(("127.0.0.1",16696)); s.sendall(("EPHEMERAL\r\nNICK "+sys.argv[1]+"\r\nUSER u 0 * :u\r\n").encode()); time.sleep(0.3)
s.sendall(("PRIVMSG #lobby :"+sys.argv[2]+"\r\n").encode()); time.sleep(0.3); s.close()
PY
}
presence() { $S/binws/airc directory --addr $AIRC --nick auditor --who carol --json | python3 -c "import json,sys; d=json.load(sys.stdin); print(json.dumps(d.get('cards', d))[:300])"; }
WILL='{"version":1,"request_id":"w","operation":"presence.set","presence":{"state":"away","ttl_seconds":600,"note":"MQTT link lost"}}'
echo "== mosquitto_sub as carol with a presence will, then SIGKILL"
mosquitto_sub -V 5 -h 127.0.0.1 -p $MQTT -i will-carol -q 1 -t airc/room/lobby \
  -D connect user-property airc-nick carol -D connect user-property airc-token $TOKEN \
  --will-topic airc/requests --will-qos 1 --will-payload "$WILL" > /dev/null 2>&1 &
SUB=$!; sleep 1
echo "before:"; presence
kill -9 $SUB; wait $SUB 2>/dev/null; sleep 1
echo "after SIGKILL:"; presence
CLEAR='{"version":1,"request_id":"c1","operation":"presence.set","presence":{"state":"clear","ttl_seconds":0}}'
clear() { mosquitto_pub -V 5 -h 127.0.0.1 -p $MQTT -q 1 -t airc/requests -i clear-carol -D connect user-property airc-nick carol -D connect user-property airc-token $TOKEN -m "$CLEAR"; echo "cleared:"; presence; }
echo "== mosquitto_sub with the will, exiting after one message: DISCONNECT 0x00, no will"
clear
mosquitto_sub -V 5 -h 127.0.0.1 -p $MQTT -i will-carol -q 1 -t airc/room/lobby -C 1   -D connect user-property airc-nick carol -D connect user-property airc-token $TOKEN   --will-topic airc/requests --will-qos 1 --will-payload "$WILL" > /dev/null 2>&1 &
SUB=$!; sleep 1
native will-poster "one message for mosquitto_sub"
wait $SUB; sleep 1
echo "after a clean exit:"; presence
echo "== mosquitto_sub with the will, then SIGINT: mosquitto_sub sends DISCONNECT 0x04, asking for the will"
clear
mosquitto_sub -V 5 -h 127.0.0.1 -p $MQTT -i will-carol -q 1 -t airc/room/lobby   -D connect user-property airc-nick carol -D connect user-property airc-token $TOKEN   --will-topic airc/requests --will-qos 1 --will-payload "$WILL" > /dev/null 2>&1 &
SUB=$!; sleep 1
kill -INT $SUB; wait $SUB 2>/dev/null; sleep 1
echo "after SIGINT:"; presence
echo "== paho-mqtt over WebSocket"
python3 $R/paho_ws.py 127.0.0.1 $MQTTWS paho-ws $AIRC
echo "== MQTT.js 5.16.0 in headless Chromium"
(cd $R && node browser.mjs "http://127.0.0.1:$PAGE/browser.html?url=ws://127.0.0.1:$MQTTWS/mqtt&nick=browser-guest") > $R/state/browser.out 2>&1 &
BROWSER=$!; sleep 4
native browser-poster "to the browser"
wait $BROWSER; cat $R/state/browser.out
echo "== native history"
python3 - <<PY
import socket,time,json,base64
s=socket.create_connection(("127.0.0.1",16696)); s.sendall(b"EPHEMERAL\r\nNICK auditor2\r\nUSER u 0 * :u\r\n"); time.sleep(0.5); s.recv(65536)
s.sendall(b"HISTORY #lobby 20\r\n"); time.sleep(0.5)
for line in s.recv(65536).decode().split("\r\n"):
    if " 760 " in line:
        m=json.loads(line.split(" :",1)[1]); print(m["from"], base64.b64decode(m["body_base64"]+"==").decode())
PY
kill $PAGESERVER $GATEWAY $AIRCD 2>/dev/null; wait 2>/dev/null
