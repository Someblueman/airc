export TMPDIR=$B/tmp
S=/home/user/airc-bench/scripts/binary-size.sh
for c in 9d37a96 18c00f8 0c07095 1307953 94bfd1a; do
  d=$B/wt-$c; [ $c = 94bfd1a ] && d=/home/user/airc-bench
  $S -C $d --target linux/amd64 --target darwin/arm64 --target linux/arm64 > $B/size/$c.txt 2>$B/size/$c.err
