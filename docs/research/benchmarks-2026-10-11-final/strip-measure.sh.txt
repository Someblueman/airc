#!/bin/bash
B=/tmp/claude-0/-home-user-airc/c3f35c0f-2deb-5898-b662-f7faf329fb28/scratchpad/bench2
export TMPDIR=$B/tmp
for g in none grpc webrtc quic coap mcp webrtc,quic,coap grpc,webrtc,quic,coap,mcp; do
  $B/tools/strip.sh $g >/dev/null || { echo "$g FAILED"; continue; }
  cd $B/wt-exp
  for tags in "" grpcnotrace; do
    go build -tags "$tags" -o $TMPDIR/gw ./cmd/airc-gateway; a=$(wc -c < $TMPDIR/gw)
    go build -tags "$tags" -trimpath -ldflags='-s -w' -o $TMPDIR/gw ./cmd/airc-gateway; b=$(wc -c < $TMPDIR/gw); c=$(gzip -9 -n -c $TMPDIR/gw | wc -c)
    echo "removed=$g tags=${tags:-none} installed=$a trimpath_stripped=$b trimpath_stripped_gzip=$c"
  done
done
cd $B/wt-exp && git reset -q --hard && git clean -qfd; rm -f $TMPDIR/gw
