# Scratch experiment only: drop owner entries so the linker can discard a carrier.
import sys,re
groups={
 'webrtc':['webrtc'],
 'grpc':['grpc'],
 'quic':['http3','webtransport','rawquic','rawquic_datagrams'],
 'coap':['coap_tcp','coaps_tcp','coap_udp','dtls','coaps','coap_oscore'],
 'mcp':['mcp_http'],
}
sel=sys.argv[1].split(',')
p='internal/gateway/service/owners.go'; s=open(p).read()
for g in sel:
    for k in groups[g]:
        s,n=re.subn(r'\n\t"%s":\s+[^\n]*'%re.escape(k),'',s); assert n==1,(k,n)
open(p,'w').write(s)
if 'coap' in sel:
    p='internal/gateway/service/author_packet.go'; s=open(p).read()
    s=s.replace('if l.Protocol == "coap_udp" {','if false && l.Protocol == "coap_udp" {'); open(p,'w').write(s)
