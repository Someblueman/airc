package server

import (
	"encoding/json"
	"fmt"
	"github.com/Someblueman/airc/internal/protocol"
	"strings"
	"testing"
)

func probeRing(n int, body string) historyRing {
	h := newHistory(n)
	for i := 0; i < n; i++ {
		h.add(Message{ID: fmt.Sprintf("%032x", i+1), From: "writer", Target: fmt.Sprintf("#r%d", i%8), Body: body})
	}
	return h
}
func BenchmarkRoadmapAppend(b *testing.B) {
	for _, n := range []int{1000, 10000} {
		for _, quota := range []bool{false, true} {
			b.Run(fmt.Sprintf("n%d/quota%t", n, quota), func(b *testing.B) {
				h := probeRing(n, "hello @worker")
				if quota {
					h.quotas = map[string]roomSettings{"#r0": {HistoryLimit: n / 4}}
				}
				b.ReportAllocs()
				b.ResetTimer()
				for i := 0; i < b.N; i++ {
					h.add(Message{ID: fmt.Sprintf("new%029x", i), From: "writer", Target: fmt.Sprintf("#r%d", i%8), Body: "hello @worker"})
				}
			})
		}
	}
}
func BenchmarkRoadmapCheck(b *testing.B) {
	for _, n := range []int{1000, 10000} {
		for _, targets := range []int{1, 16, 64} {
			b.Run(fmt.Sprintf("n%d/targets%d", n, targets), func(b *testing.B) {
				s := New(Config{HistoryLimit: n})
				s.history = probeRing(n, "ordinary message")
				r := protocol.CheckRequest{MaxMessages: 100}
				for i := 0; i < targets; i++ {
					r.Targets = append(r.Targets, protocol.CheckTarget{Target: fmt.Sprintf("#absent%d", i), After: s.history.at(0).ID, Limit: 100})
				}
				raw, _ := json.Marshal(r)
				actor := &session{client: Client{Nick: "reader"}, out: make(chan string, 2048), done: make(chan struct{})}
				b.ReportAllocs()
				b.ResetTimer()
				for i := 0; i < b.N; i++ {
					s.checkLocked(actor, protocol.Command{Trailing: string(raw)})
					for len(actor.out) > 0 {
						<-actor.out
					}
					actor.outBytes.Store(0)
				}
			})
		}
	}
}
func BenchmarkRoadmapSearch(b *testing.B) {
	for _, size := range []int{80, 4096} {
		b.Run(fmt.Sprintf("body%d", size), func(b *testing.B) {
			h := probeRing(10000, strings.Repeat("AbCdEfGh", size/8))
			b.ReportAllocs()
			b.ResetTimer()
			for i := 0; i < b.N; i++ {
				h.search("*", "*", "not-found", "*", 50)
			}
		})
	}
}
