func BenchmarkRoadmapUI(b *testing.B) {
	for _, size := range []int{80, 1024} {
		for _, dirty := range []bool{false, true} {
			b.Run(fmt.Sprintf("body%d/dirty%t", size, dirty), func(b *testing.B) {
				m := newUIModel("human", []string{"#room"}, time.Now)
				m.width, m.height = 160, 45
				for i := 0; i < 500; i++ {
					m.route(&irc.MessageEvent{ID: fmt.Sprint(i), From: fmt.Sprintf("agent%d", i%20), Target: "#room", Message: strings.Repeat("content ", size/8), Timestamp: time.Now()}, true)
				}
				m.view()
				b.ReportAllocs()
				b.ResetTimer()
				for i := 0; i < b.N; i++ {
					if dirty {
						m.cur().version++
					}
					m.view()
				}
			})
		}
	}
}
func TestRoadmapInputProbe(t *testing.T) {
	paste := parseKeys([]byte("\x1b[200~first line\nsecond line\x1b[201~"))
	enter := 0
	for _, k := range paste {
		if k.kind == keyEnter {
			enter++
		}
	}
	whole := parseKeys([]byte("🙂"))
	split := append(parseKeys([]byte{0xf0, 0x9f}), parseKeys([]byte{0x99, 0x82})...)
	t.Logf("bracketed paste Enter count=%d; whole emoji keys=%d; split emoji keys=%d", enter, len(whole), len(split))
}
func TestRoadmapContextProbe(t *testing.T) {
	agentEnv(t)
	address := cliTestServerWith(t, server.Config{HistoryLimit: 1100, OutboundBytes: 16 << 20})
	c, err := irc.Dial(irc.Config{Nick: "writer", Addr: address, Ephemeral: true})
	if err != nil {
		t.Fatal(err)
	}
	defer c.Close()
	ctx, cancel := context.WithTimeout(context.Background(), 30*time.Second)
	defer cancel()
	if err := c.Send("#room", "question"); err != nil {
		t.Fatal(err)
	}
	root, err := awaitSend(ctx, c, "writer", "#room", "question", "", "", "")
	if err != nil {
		t.Fatal(err)
	}
	body := strings.Repeat("Evidence with Mixed CASE and code. ", 100)
	for i := 0; i < 999; i++ {
		if err := c.Reply(root.ID, body); err != nil {
			t.Fatal(err)
		}
		if _, err := awaitSend(ctx, c, "writer", "#room", body, root.ID, "", ""); err != nil {
			t.Fatal(err)
		}
	}
	for _, limit := range []int{50, 500, 1000} {
		for repeat := 0; repeat < 2; repeat++ {
			runtime.GC()
			var before, after runtime.MemStats
			runtime.ReadMemStats(&before)
			var out, diagnostics bytes.Buffer
			start := time.Now()
			err := runContext([]string{root.ID, "--addr", address, "--nick", "reader", "--limit", fmt.Sprint(limit), "--max-bytes", "32768", "--json"}, &out, &diagnostics)
			elapsed := time.Since(start)
			runtime.ReadMemStats(&after)
			if err != nil {
				t.Fatal(err, diagnostics.String())
			}
			t.Logf("context limit=%d repeat=%d elapsed_ms=%.3f allocated_bytes=%d output_bytes=%d", limit, repeat, float64(elapsed)/float64(time.Millisecond), after.TotalAlloc-before.TotalAlloc, out.Len())
		}
	}
}
