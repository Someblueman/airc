func zzSilent(t *testing.T, address string) {
	var l net.Listener
	var err error
	for range 50 {
		if l, err = net.Listen("tcp", address); err == nil {
			break
		}
		time.Sleep(20 * time.Millisecond)
	}
	if err != nil {
		t.Fatal(err)
	}
	var mu sync.Mutex
	var conns []net.Conn
	go func() {
		for {
			c, err := l.Accept()
			if err != nil {
				return
			}
			mu.Lock()
			conns = append(conns, c)
			mu.Unlock()
		}
	}()
	t.Cleanup(func() {
		_ = l.Close()
		mu.Lock()
		for _, c := range conns {
			_ = c.Close()
		}
		mu.Unlock()
	})
}

func zzForget() {
	_ = filepath.WalkDir(os.Getenv("AIRC_STATE_DIR"), func(p string, d fs.DirEntry, err error) error {
		if err == nil && !d.IsDir() && strings.HasPrefix(d.Name(), "last-") {
			_ = os.Remove(p)
		}
		return nil
	})
}

func TestZZMeasure(t *testing.T) {
	out, err := os.OpenFile(os.Getenv("ZZ_OUT"), os.O_APPEND|os.O_CREATE|os.O_WRONLY, 0o644)
	if err != nil {
		t.Fatal(err)
	}
	defer out.Close()
	tree := os.Getenv("ZZ_TREE")
	record := func(scenario string, d time.Duration, err error) {
		status := "ok"
		if err != nil {
			status = "err"
		}
		fmt.Fprintf(out, "%s %s %.1f %s\n", tree, scenario, float64(d.Microseconds())/1000, status)
	}
	timed := func(scenario string, fn func() error) {
		start := time.Now()
		err := fn()
		record(scenario, time.Since(start), err)
	}
	const n = 8
	send := func(common []string, body string) func() error {
		return func() error {
			_, _, err := sendJSON(t, append(common, "--message", body)...)
			return err
		}
	}
	history := func(addr string) func() error {
		return func() error {
			return run([]string{"history", "#ops", "--addr", addr, "--json"}, nil, io.Discard, io.Discard)
		}
	}
	t.Run("healthy", func(t *testing.T) {
		f := newFallbackFixture(t)
		common := []string{"--addr", f.proxy.addr(), "--nick", "alice", "--channel", "#ops"}
		_ = send(common, "learn")()
		for i := range n {
			timed("send_healthy", send(common, fmt.Sprint("h", i)))
			timed("history_healthy", history(f.proxy.addr()))
		}
	})
	t.Run("refused", func(t *testing.T) {
		f := newFallbackFixture(t)
		common := []string{"--addr", f.proxy.addr(), "--nick", "alice", "--channel", "#ops"}
		_ = send(common, "learn")()
		f.proxy.close()
		for i := range n {
			zzForget()
			timed("send_refused", send(common, fmt.Sprint("r", i)))
			zzForget()
			timed("history_refused", history(f.proxy.addr()))
		}
	})
	t.Run("silent", func(t *testing.T) {
		f := newFallbackFixture(t)
		common := []string{"--addr", f.proxy.addr(), "--nick", "alice", "--channel", "#ops"}
		_ = send(common, "learn")()
		f.proxy.close()
		zzSilent(t, f.proxy.addr())
		for i := range n {
			zzForget()
			timed("send_silent_cold", send(common, fmt.Sprint("s", i)))
			zzForget()
			timed("history_silent_cold", history(f.proxy.addr()))
		}
		_ = send(common, "prime")()
		for i := range n {
			timed("send_silent_warm", send(common, fmt.Sprint("w", i)))
			timed("history_silent_warm", history(f.proxy.addr()))
		}
	})
	t.Run("mid_send_silence", func(t *testing.T) {
		for i := range 4 {
			f := newFallbackFixture(t)
			common := []string{"--addr", f.proxy.addr(), "--nick", "alice", "--channel", "#ops"}
			_ = send(common, "learn")()
			f.proxy.setBlackhole()
			timed("send_goes_silent_mid_post", send(common, fmt.Sprint("m", i)))
		}
	})
}
