#!/bin/bash
# usage: round.sh OUTDIR ROUNDS  -- runs every benchmark binary interleaved base/head
S=/tmp/claude-0/-home-user-airc/c3f35c0f-2deb-5898-b662-f7faf329fb28/scratchpad/bench
OUT=$1; N=$2
mkdir -p $OUT
export TMPDIR=$S/tmp
for r in $(seq 1 $N); do
  for t in base head; do
    for p in internal_server pkg_irc internal_gateway_backend cmd_airc; do
      dir=/home/user/airc-bench-$t/$(echo $p | tr _ / | sed 's|internal/gateway/backend|internal/gateway/backend|')
      case $p in
        internal_server) dir=/home/user/airc-bench-$t/internal/server; pat='^Benchmark(History|HistoryAddWithQuota|ContextWireBudget)$';;
        pkg_irc) dir=/home/user/airc-bench-$t/pkg/irc; pat='^BenchmarkConnect$';;
        internal_gateway_backend) dir=/home/user/airc-bench-$t/internal/gateway/backend; pat='.';;
        cmd_airc) dir=/home/user/airc-bench-$t/cmd/airc; pat='^Benchmark(UIAppendRedraw|AgentCLI)$';;
      esac
      (cd $dir && AIRC_BENCH_BINARY=$S/bin/$t.airc $S/bin/$t.$p.test -test.run '^$' -test.bench "$pat" -test.count 1 -test.benchmem) >> $OUT/$t.txt 2>&1
    done
  done
  echo "round $r done $(date +%T) load $(cut -d' ' -f1 /proc/loadavg)" >> $OUT/progress
done
