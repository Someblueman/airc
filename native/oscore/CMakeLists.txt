cmake_minimum_required(VERSION 3.20)
project(airc_oscore C)

if(NOT IS_ABSOLUTE "${LIBCOAP_SOURCE}")
  message(FATAL_ERROR "LIBCOAP_SOURCE must name the pinned, patched source directory")
endif()

# The dependency remains isolated from the Go build. No install step is needed.
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ENABLE_OSCORE ON CACHE BOOL "" FORCE)
set(ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(ENABLE_EXAMPLES ON CACHE BOOL "" FORCE)
set(ENABLE_DOCS OFF CACHE BOOL "" FORCE)
set(DTLS_BACKEND gnutls CACHE STRING "" FORCE)
set(WITH_GNUTLS ON CACHE BOOL "" FORCE)
set(WITH_OPENSSL OFF CACHE BOOL "" FORCE)
set(WITH_MBEDTLS OFF CACHE BOOL "" FORCE)
set(WITH_TINYDTLS OFF CACHE BOOL "" FORCE)
set(WITH_WOLFSSL OFF CACHE BOOL "" FORCE)
option(ENABLE_SANITIZER "Run native code with ASan and UBSan" ON)
add_subdirectory("${LIBCOAP_SOURCE}" libcoap)

enable_testing()
add_executable(oscore-persistence-check persistence_check.c)
add_executable(oscore-context-check context_check.c)
add_executable(oscore-persistence-probe persistence_probe.c persistence.c)
foreach(target oscore-persistence-check oscore-context-check oscore-persistence-probe)
  target_compile_features(${target} PRIVATE c_std_11)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -pedantic)
  target_link_libraries(${target} PRIVATE coap-3)
  if(ENABLE_SANITIZER)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
endforeach()
add_test(NAME oscore-persistence-boundaries COMMAND oscore-persistence-check)
set_tests_properties(oscore-persistence-boundaries PROPERTIES TIMEOUT 20)
